# Ubuntu (same flag as the LIBERO benchmark deps) because they pin
# SAPIEN/mplib versions that are validated for RoboTwin 2.0, not for the
# rest of the workspace.
#
# NOT installable from this file:
#   nvidia-curobo==0.7.8 — the PyPI `nvidia-curobo` is a 1 KB placeholder
#   stub, not NVlabs/curobo. Install the real v1 source build (RoboTwin uses
#   the v1 API; v2/main renamed everything):
#     sudo apt install cuda-toolkit-12-8   # prerequisite, host CUDA toolkit
#     pip install --no-build-isolation \
#       "git+https://github.com/NVlabs/curobo.git@v0.7.8"
#
# After installing curobo, materialize the embodiment planner configs (cuRobo
# v0.7.8 does not expand ${ASSETS_PATH} in YAML):
#   cp libs/robotwin/assets/embodiments/aloha-agilex/curobo_left_tmp.yml \
#         libs/robotwin/assets/embodiments/aloha-agilex/curobo_left.yml
#   cp libs/robotwin/assets/embodiments/aloha-agilex/curobo_right_tmp.yml \
#         libs/robotwin/assets/embodiments/aloha-agilex/curobo_right.yml
#   sed -i "s|\${ASSETS_PATH}|<abs path to libs/robotwin>|g" \
#         libs/robotwin/assets/embodiments/aloha-agilex/curobo_{left,right}.yml
#
# The libs/robotwin submodule clones from GitHub on --with-benchmark hosts;
# networks without GitHub access need a mirror or manual clone first.
